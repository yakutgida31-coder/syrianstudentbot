class LinkParser(HTMLParser):

    def __init__(self):
        super().__init__()
        self.links = []
        self.current_href = None
        self.current_text = []
        self.in_a = False

    def handle_starttag(self, tag, attrs):

        if tag.lower() != "a":
            return

        attributes = dict(attrs)

        href = attributes.get("href")

        if href:
            self.current_href = href
            self.current_text = []
            self.in_a = True

    def handle_data(self, data):

        if self.in_a:
            self.current_text.append(data)

    def handle_endtag(self, tag):

        if tag.lower() != "a":
            return

        if self.in_a and self.current_href:

            text = " ".join(self.current_text)
            text = " ".join(text.split())

            if text:

                self.links.append({
                    "title": text,
                    "link": self.current_href
                })

        self.current_href = None
        self.current_text = []
        self.in_a = False


def download_page(url):

    request = urllib.request.Request(
        url,
        headers={
            "User-Agent": "Mozilla/5.0"
        }
    )

    with urllib.request.urlopen(request, timeout=20) as response:

        data = response.read()

        charset = response.headers.get_content_charset()

        if not charset:
            charset = "utf-8"

        return data.decode(charset, errors="ignore")


def make_absolute_url(base_url, link):

    return urllib.parse.urljoin(base_url, link)


# =========================================================
# جلب آخر خبر من سانا
# =========================================================

async def get_sana_news():

    try:

        html = await asyncio.to_thread(
            download_page,
            SANA_URL
        )

        parser = LinkParser()
        parser.feed(html)

        for item in parser.links:

            title = item["title"].strip()
            link = make_absolute_url(
                SANA_URL,
                item["link"]
            )

            if not title:
                continue

            if not link.startswith("http"):
                continue

            # نفضل روابط الأخبار وليس روابط الأقسام
            if "sana.sy" not in link:
                continue

            if link.rstrip("/") == SANA_URL.rstrip("/"):
                continue

            return {
                "source": "سانا",
                "title": title,
                "link": link
            }

    except Exception as e:

        print("خطأ في جلب أخبار سانا:", e)

    return None


# =========================================================
# جلب آخر خبر من قناة وزارة التعليم العالي

async def get_ministry_news():

    try:

        html = await asyncio.to_thread(
            download_page,
            MINISTRY_URL
        )

        parser = LinkParser()
        parser.feed(html)

        possible = []

        for item in parser.links:

            title = item["title"].strip()
            link = make_absolute_url(
                MINISTRY_URL,
                item["link"]
            )

            if not title:
                continue

            if "t.me" not in link:
                continue

            possible.append({
                "source": "وزارة التعليم العالي",
                "title": title,
                "link": link
            })

        if possible:

            # نأخذ آخر رابط منشور في الصفحة
            return possible[-1]

    except Exception as e:

        print("خطأ في جلب أخبار وزارة التعليم العالي:", e)

    return None

async def show_universities(query):

    keyboard = []

    for i, university in enumerate(UNIVERSITIES):

        if university["type"] == "حكومية":
            icon = "🏛"
        else:
            icon = "🏫"

        keyboard.append([
            InlineKeyboardButton(
                f"{icon} {university['name']}",
                callback_data=f"uni_{i}"
            )
        ])

    keyboard.append([
        InlineKeyboardButton(
            "🏠 الرئيسية",
            callback_data="home"
        )
    ])

    await query.edit_message_text(
        "🏛 الجامعات السورية\n\nاختر الجامعة:",
        reply_markup=InlineKeyboardMarkup(keyboard)
    )


async def show_university(query, index):

    if index < 0 or index >= len(UNIVERSITIES):
        return

    university = UNIVERSITIES[index]

    text = (
        f"🎓 {university['name']}\n\n"
        f"🏷 النوع: {university['type']}\n"
        f"📍 الموقع: {university['city']}\n\n"
        f"📚 عدد الاختصاصات: "
        f"{len(university['faculties'])}"
    )

    keyboard = [

        [
            InlineKeyboardButton(
                "📚 الكليات والاختصاصات",
                callback_data=f"faculties_{index}"
            )
        ],

        [
            InlineKeyboardButton(
                "⬅️ الجامعات",
                callback_data="universities"
            ),

            InlineKeyboardButton(
                "🏠 الرئيسية",
                callback_data="home"
            )
        ]
    ]

    await query.edit_message_text(
        text,
        reply_markup=InlineKeyboardMarkup(keyboard)
    )


async def show_faculties(query, university_index):

    if university_index < 0 or university_index >= len(UNIVERSITIES):
        return

    university = UNIVERSITIES[university_index]

    keyboard = []

    for i, faculty in enumerate(
        university["faculties"]
    ):

        keyboard.append([
            InlineKeyboardButton(
                f"📘 {faculty}",
                callback_data=f"faculty_{university_index}_{i}"
            )
        ])

    keyboard.append([
        InlineKeyboardButton(
            "⬅️ الجامعة",
            callback_data=f"uni_{university_index}"
        )
    ])

    await query.edit_message_text(
        f"📚 اختصاصات {university['name']}\n\n"
        "اختر الاختصاص:",
        reply_markup=InlineKeyboardMarkup(keyboard)
    )


async def show_faculty(
    query,
    university_index,
    faculty_index
):

    if university_index < 0 or university_index >= len(UNIVERSITIES):
        return

    university = UNIVERSITIES[university_index]

    if faculty_index < 0 or faculty_index >= len(
        university["faculties"]
    ):
        return

    faculty = university["faculties"][faculty_index]

    text = (
        f"🎓 {university['name']}\n\n"
        f"📘 الاختصاص:\n{faculty}\n\n"
        f"📍 {university['city']}\n"
        f"🏷 {university['type']}"
    )

    keyboard = [

        [
            InlineKeyboardButton(
                "⬅️ الاختصاصات",
                callback_data=f"faculties_{university_index}"
            )
        ],

        [
            InlineKeyboardButton(
                "🏠 الرئيسية",
                callback_data="home"
            )
        ]
    ]

    await query.edit_message_text(
        text,
        reply_markup=InlineKeyboardMarkup(keyboard)
    )

def parse_number(value):

    value = value.replace(",", "")
    value = value.replace(" ", "")

    try:
        return float(value)
    except ValueError:
        return None


def calculate_range(
    price_text,
    hours
):

    price_text = price_text.replace(
        " ",
        ""
    )

    if "-" in price_text:

        parts = price_text.split("-")

        if len(parts) == 2:

            first = parse_number(
                parts[0]
            )

            second = parse_number(
                parts[1]
            )

            if first is not None and second is not None:

                return (
                    first * hours,
                    second * hours
                )

    price = parse_number(price_text)

    if price is not None:

        total = price * hours

        return total, total

    return None

async def calculate_semester(
    query,

    specialty_index,

):

        university_index
    )

    if university_name is None:
        return

    specialties = list(
        TUITION[university_name].keys()
    )

    if specialty_index < 0 or specialty_index >= len(
        specialties
    ):
        return

    specialty = specialties[specialty_index]

    price, total_hours = TUITION[
        university_name
    ][specialty]

    result = calculate_range(
        price,
        hours
    )

    text = (
        f"🏫 {university_name}\n\n"
        f"📚 {specialty}\n\n"
        f"💵 سعر الساعة: {price} ل.س\n"

    )

        text += (
            "⚠️ سعر الساعة مسجل بصيغة "
            "غير قابلة للحساب بشكل واحد.\n\n"
            "تم عرض السعر كما ورد في البيانات "
            "بدون اختراع قيمة."
        )

        minimum, maximum = result

        if minimum == maximum:

            text += (
                "💰 القسط المحسوب:\n"
                f"{minimum:,.0f} ل.س"
            )

        else:

            text += (
                "💰 القسط المتوقع:\n"
                f"من {minimum:,.0f} "
                f"إلى {maximum:,.0f} ل.س"
            )

    keyboard = [

        [
            InlineKeyboardButton(
                "🔢 تغيير عدد الساعات",
                callback_data=(
                    f"calcspec_"
                    f"{university_index}_"
                    f"{specialty_index}"

            InlineKeyboardButton(
                "🏫 تغيير الجامعة",
                callback_data="calculator"
            )
        ],

        [
            InlineKeyboardButton(
                "🏠 الرئيسية",
                callback_data="home"
            )
        ]
    ]

    await query.edit_message_text(
        text,
        reply_markup=InlineKeyboardMarkup(keyboard)
    )
